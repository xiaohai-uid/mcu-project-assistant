/* 智能浇花系统 — Arduino UNO */
#include <DHT.h>
#include <LiquidCrystal_I2C.h>

// 引脚定义
const int PIN_SOIL_MOISTURE = A0;
const int PIN_DHT22 = 2;
const int PIN_RELAY = 3;

// 全局对象
DHT dht(PIN_DHT22, DHT22);
LiquidCrystal_I2C lcd(0x27, 16, 2);

const int MOISTURE_THRESHOLD = 400;

void setup() {
  Serial.begin(9600);
  dht.begin();
  pinMode(PIN_RELAY, OUTPUT);
  lcd.init(); lcd.backlight();
}

void loop() {
  int moisture = analogRead(PIN_SOIL_MOISTURE);
  float temp = dht.readTemperature();

  if (moisture < MOISTURE_THRESHOLD) {
    digitalWrite(PIN_RELAY, HIGH); // 浇水
  } else {
    digitalWrite(PIN_RELAY, LOW);
  }
  delay(2000);
}

🚀 从零到运行的完整步骤

  • 1下载安装 Arduino IDE
  • 2用 USB 数据线连接 Arduino UNO 到电脑
  • 3IDE → 工具 → 开发板 → 选择 "Arduino Uno"
  • 4IDE → 工具 → 端口 → 选择出现的 COM 口
  • 5安装库：工具 → 管理库 → 搜索 "DHT sensor library", "LiquidCrystal_I2C"
  • 6将代码复制到 IDE 编辑器
  • 7点击 "上传" (→)，等待 "Done uploading"
  • 8打开串口监视器 (9600 baud) 查看运行状态
